\section{Governing equation}
\begin{equation}
\frac{\partial c}{\partial t}
= D\left(\frac{\partial^2 c}{\partial r^2}+\frac{2}{r}\frac{\partial c}{\partial r}\right)+S_0
=\frac{D}{r^2}\frac{\partial}{\partial r}\left(r^2\frac{\partial c}{\partial r}\right)+S_0.
\end{equation}
\begin{align}
c(r,0)&=0,\\
\frac{\partial c}{\partial r}(0,t)&=0,\\
-D\frac{\partial c}{\partial r}(R,t)&=h\,c(R,t),\\
\Bi&=\frac{hR}{D}.
\end{align}

\section{Sturm--Liouville form}
\begin{equation}
\frac{\dd}{\dd r}\left(p\phi'\right)+q\phi+\lambda\sigma\phi=0,
\qquad p=r^2,\quad q=0,\quad \sigma=r^2.
\end{equation}

\section{Steady-state split}
\begin{equation}
c(r,t)=w(r)+v(r,t).
\end{equation}
\begin{equation}
\frac{D}{r^2}(r^2w')'+S_0=0.
\end{equation}
\begin{equation}
(r^2w')'=-\frac{S_0r^2}{D},\qquad
r^2w'=-\frac{S_0r^3}{3D},\qquad
w'=-\frac{S_0r}{3D}.
\end{equation}
\begin{equation}
\boxed{w(r)=\frac{S_0R}{3h}+\frac{S_0}{6D}(R^2-r^2)}.
\end{equation}
\begin{equation}
v_t=\frac{D}{r^2}(r^2v_r)_r,\qquad
v_r(0,t)=0,\qquad -Dv_r(R,t)=hv(R,t),\qquad v(r,0)=-w(r).
\end{equation}

\section{Eigenvalue problem}
\begin{equation}
\frac{\dd}{\dd r}\left(r^2\frac{\dd\phi}{\dd r}\right)+\lambda r^2\phi=0.
\end{equation}
\begin{equation}
u=r\phi,\qquad u''+\lambda u=0.
\end{equation}
\begin{equation}
\phi(r)=\frac{\sin(\sqrt{\lambda}\,r)}{r}.
\end{equation}
\begin{equation}
\mu=\sqrt{\lambda}R,\qquad
\boxed{\mu_n\cot\mu_n=1-\Bi},\qquad
\boxed{\lambda_n=\frac{\mu_n^2}{R^2}}.
\end{equation}
\begin{equation}
\Bi\ll1:\quad \mu_1\approx\sqrt{3\Bi},
\qquad
\Bi\to\infty:\quad \mu_n\to n\pi.
\end{equation}

\section{Orthogonality and normalization}
\begin{equation}
\int_0^R\phi_n\phi_m r^2\dd r=0,\qquad n\ne m.
\end{equation}
\begin{equation}
N_n=\int_0^R\phi_n^2r^2\dd r
=\frac{R}{2}\left[1-\frac{\sin(2\mu_n)}{2\mu_n}\right]
=\frac{R}{2}\left[1-(1-\Bi)\frac{\sin^2\mu_n}{\mu_n^2}\right].
\end{equation}

\section{Expansion coefficients}
\begin{equation}
a_n=-\frac{1}{N_n}\int_0^Rw(r)r\sin(k_nr)\dd r,
\qquad k_n=\frac{\mu_n}{R}.
\end{equation}
\begin{align}
I_1&=\int_0^Rr\sin(kr)\dd r
=\frac{\Bi\sin\mu}{k^2},\\
I_3&=\int_0^Rr^3\sin(kr)\dd r
=\sin\mu\left[(2+\Bi)\frac{R^2}{k^2}-\frac{6\Bi}{k^4}\right].
\end{align}
\begin{equation}
\boxed{a_n=-\frac{\Bi S_0R^4\sin\mu_n}{D\mu_n^4N_n}}.
\end{equation}

\section{Source active: $0\le t\le t_c$}
\begin{equation}
\boxed{
c(r,t)=\frac{S_0R}{3h}+\frac{S_0}{6D}(R^2-r^2)
+\sum_{n\ge1}a_n\frac{\sin(\mu_nr/R)}{r}
\exp\left(-\frac{D\mu_n^2t}{R^2}\right)}.
\end{equation}
\begin{equation}
\tau_1=\frac{R^2}{D\mu_1^2}.
\end{equation}
\begin{equation}
\dot m(t)=4\pi R^2h\,c(R,t).
\end{equation}

\section{After source switch-off: $t>t_c$}
\begin{equation}
c(r,t)=\sum_{n\ge1}b_n\frac{\sin(\mu_nr/R)}{r}
\exp\left[-\frac{D\mu_n^2(t-t_c)}{R^2}\right].
\end{equation}
\begin{equation}
\boxed{b_n=a_n\left[\exp\left(-\frac{D\mu_n^2t_c}{R^2}\right)-1\right]}.
\end{equation}

\section{Five-layer TRISO extension}
For layer $i$:
\begin{equation}
\frac{\partial c}{\partial t}
=\frac{D_i}{r^2}\frac{\partial}{\partial r}\left(r^2\frac{\partial c}{\partial r}\right)+S_i.
\end{equation}
\begin{equation}
S_i=\begin{cases}S_0,&i=1,\\0,&i=2,\ldots,5.\end{cases}
\end{equation}
\begin{equation}
c(r_i^-,t)=c(r_i^+,t),\qquad
D_i\frac{\partial c}{\partial r}\bigg|_{r_i^-}
=D_{i+1}\frac{\partial c}{\partial r}\bigg|_{r_i^+}.
\end{equation}
\begin{equation}
\frac{\dd}{\dd r}\left(D(r)r^2\frac{\dd\phi}{\dd r}\right)+\lambda r^2\phi=0,
\qquad p(r)=D(r)r^2,\quad \sigma(r)=r^2.
\end{equation}
\begin{equation}
k_i=\sqrt{\frac{\lambda}{D_i}},\qquad
u^{(i)}(r)=\alpha_i\sin(k_ir)+\beta_i\cos(k_ir),\qquad \beta_1=0.
\end{equation}
\begin{equation}
\boxed{\det M(\lambda)=0}.
\end{equation}
